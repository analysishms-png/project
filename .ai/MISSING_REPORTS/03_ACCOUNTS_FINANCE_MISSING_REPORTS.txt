REPORT 1: BankBook
============================================================
HMS.bas FORM NAME: BankBook
HMS.bas LINE: 1888
BUSINESS PURPOSE: Bank book showing all bank transactions
  with opening balance, receipts, payments, closing balance.

HMS.bas SQL LOGIC:
  SELECT a.Code AS AcCode, a.Name AS AcName,
    a.OpBal,
    ISNULL(SUM(CASE WHEN t.DrCr = 'D' THEN t.Amount END), 0) AS TotalDr,
    ISNULL(SUM(CASE WHEN t.DrCr = 'C' THEN t.Amount END), 0) AS TotalCr,
    a.OpBal + ISNULL(SUM(CASE WHEN t.DrCr = 'D' THEN t.Amount END), 0)
    - ISNULL(SUM(CASE WHEN t.DrCr = 'C' THEN t.Amount END), 0) AS ClBal
  FROM AcGroup a
  LEFT JOIN AcCntrlTrn t ON a.Code = t.AcCode
    AND t.Vdate BETWEEN @fromdate AND @todate
  WHERE a.BankYN = 'Y'
  AND a.LogSite_Code = @site
  GROUP BY a.Code, a.Name, a.OpBal
  ORDER BY a.Name

REPORT 3: CompanyAnalysis (Company Revenue Analysis)
============================================================
HMS.bas FORM NAME: CompanyAnalysis
HMS.bas LINE: 511
BUSINESS PURPOSE: Revenue analysis by company showing room
  nights, charges, payments, outstanding.

HMS.bas SQL LOGIC:
  SELECT sg.Name AS CompanyName, sg.SubCode,
    COUNT(DISTINCT r.DocId) AS RoomNights,
    SUM(pc.AmtDr) AS TotalCharges,
    SUM(pc.AmtCr) AS TotalPayments,
    SUM(pc.AmtDr) - SUM(pc.AmtCr) AS Outstanding,
    CASE WHEN COUNT(DISTINCT r.DocId) > 0
      THEN SUM(pc.AmtDr) / COUNT(DISTINCT r.DocId)
      ELSE 0 END AS AvgRevenuePerRoom
  FROM SubGroup sg
  INNER JOIN GuestFolio gf ON sg.SubCode = gf.Company
  INNER JOIN RoomOcc r ON gf.DocId = r.DocId
  LEFT JOIN PayCharge pc ON gf.DocId = pc.FolioNoDocId
  WHERE sg.AcType = 'Sundry Debtors'
  AND r.ChkInDate BETWEEN @fromdate AND @todate
  AND r.PropertyID = @propertyid
  GROUP BY sg.Name, sg.SubCode
  ORDER BY Outstanding DESC

FILTERS:
  - Date From / Date To
  - Company
  - Outstanding Only

REPORT 5: DailySumm (Daily Financial Summary)
============================================================
HMS.bas FORM NAME: DailySumm
HMS.bas LINE: 1943
BUSINESS PURPOSE: Complete daily financial summary — revenue,
  payments, POS, outstanding, occupancy.

HMS.bas SQL LOGIC:
  SELECT
    -- Occupancy
    (SELECT COUNT(*) FROM RoomOcc
     WHERE Type NOT IN ('C','O')
     AND PropertyID = @pid) AS OccupiedRooms,

    -- Front Office Revenue
    (SELECT ISNULL(SUM(AmtDr),0) FROM PayCharge
     WHERE Vdate = @date AND RoomType <> 'RO'
     AND PropertyID = @pid) AS FORoomRevenue,

    -- POS Revenue
    (SELECT ISNULL(SUM(BillAmt),0) FROM Bill
     WHERE Vdate = @date AND CancelYN <> 'Y'
     AND PropertyID = @pid) AS POSRevenue,

    -- Total Payments
    (SELECT ISNULL(SUM(AmtCr),0) FROM PayCharge
     WHERE Vdate = @date AND PropertyID = @pid) AS TotalPayments,

    -- Advance Received
    (SELECT ISNULL(SUM(AmtDr),0) FROM PayCharge
     WHERE Vdate = @date AND VType IN ('ARRES','ADRES')
     AND PropertyID = @pid) AS AdvanceReceived

  - Date
