HMS.bas FORM NAME: RoomWiseRoomRevenueReport
HMS.bas LINE: 661102
BUSINESS PURPOSE: Room-wise revenue report showing each room's
  total revenue, charges, payments, tax for a period.

HMS.bas SQL LOGIC:
  SELECT r.RoomNo, rc.Name AS RoomCategory,
    r.RoomRate,
    gf.Name AS GuestName,
    SUM(pc.AmtDr) AS TotalCharges,
    SUM(pc.AmtCr) AS TotalPayments,
    SUM(pc.TaxAmt1+pc.TaxAmt2+pc.TaxAmt3) AS TotalTax,
    SUM(pc.AmtDr) - SUM(pc.AmtCr) AS Balance
  FROM RoomOcc r
  INNER JOIN GuestFolio gf ON r.DocId = gf.DocId
  INNER JOIN RoomCat rc ON r.RoomCat = rc.Code
  LEFT JOIN PayCharge pc ON gf.DocId = pc.FolioNoDocId
  WHERE r.ChkInDate <= @todate
  AND (r.ChkOutDate IS NULL OR r.ChkOutDate >= @fromdate)
  AND r.PropertyID = @propertyid
  GROUP BY r.RoomNo, rc.Name, r.RoomRate, gf.Name
  ORDER BY r.RoomNo

FILTERS:
  - Date From / Date To
  - Room Category
  - Room No
